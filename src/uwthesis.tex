% Optimized for Zotero exported BibTeX files
\DeclareSourcemap{
    \maps[datatype=bibtex]{
% if doi is present, clear issn, isbn, and url fields so they are not printed
        \map{
            \step[fieldsource=doi,final]
            \step[fieldset=issn,null]
            \step[fieldset=isbn,null]
            \step[fieldset=url,null]
        }
% fix issue with Zotero/Mendeley import improperly escaping underscores in doi field
        \map{
            \step[fieldsource=doi, match=\regexp{\{\\_\}}, replace=\regexp{_}]
        }
% Remove Zotero-specific fields that shouldn't appear in bibliography
        \map{
            \step[fieldset=keywords,null]
            \step[fieldset=file,null]
            \step[fieldset=abstract,null]
        }
% Normalize malformed Zotero/Better BibTeX initials that make biblatex enter math mode
        \map{
            \step[fieldsource=author, match=\regexp{Ladd,\s*\{\\relax\s*RS\}}, replace=\regexp{Ladd, R. S.}]
        }
% Normalize date field (Zotero may use 'date' instead of year)
        \map{
            \step[fieldsource=date,final]
            \step[fieldset=year,origfieldval]
        }
    }
}
